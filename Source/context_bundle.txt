69:    dc->ClearRenderTargetView(currentRTV, clearColor);

241:void SceneIntro::CreateRenderTarget()
242:{
243:    auto device = Graphics::Instance().GetDevice();
245:    float screenW = 1920;
246:    float screenH = 1080;
247:    if (auto window = Framework::Instance()->GetMainWindow())
248:    {
249:        screenW = static_cast<float>(window->GetWidth());
250:        screenH = static_cast<float>(window->GetHeight());
251:    }
253:    // Create Texture
254:    D3D11_TEXTURE2D_DESC textureDesc = {};
255:    textureDesc.Width = static_cast<UINT>(screenW);
256:    textureDesc.Height = static_cast<UINT>(screenH);
257:    textureDesc.MipLevels = 1;
258:    textureDesc.ArraySize = 1;
259:    textureDesc.Format = DXGI_FORMAT_R8G8B8A8_UNORM;
260:    textureDesc.SampleDesc.Count = 1;
261:    textureDesc.Usage = D3D11_USAGE_DEFAULT;
262:    textureDesc.BindFlags = D3D11_BIND_RENDER_TARGET | D3D11_BIND_SHADER_RESOURCE;
264:    device->CreateTexture2D(&textureDesc, nullptr, renderTargetTexture.ReleaseAndGetAddressOf());
265:    device->CreateRenderTargetView(renderTargetTexture.Get(), nullptr, renderTargetView.ReleaseAndGetAddressOf());
266:    device->CreateShaderResourceView(renderTargetTexture.Get(), nullptr, shaderResourceView.ReleaseAndGetAddressOf());
268:    // Create Depth Buffer
269:    textureDesc.Format = DXGI_FORMAT_D24_UNORM_S8_UINT;
270:    textureDesc.BindFlags = D3D11_BIND_DEPTH_STENCIL;
272:    device->CreateTexture2D(&textureDesc, nullptr, depthStencilTexture.ReleaseAndGetAddressOf());
273:    device->CreateDepthStencilView(depthStencilTexture.Get(), nullptr, depthStencilView.ReleaseAndGetAddressOf());
274:}
